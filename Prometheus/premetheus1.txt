# Prometheus Installation on Amazon Linux EC2
# Security Group: Open TCP port 9090 from My IP

sudo -i

useradd prometheus
usermod -aG wheel prometheus

cd /home/prometheus

wget https://github.com/prometheus/prometheus/releases/download/v2.51.1/prometheus-2.51.1.linux-amd64.tar.gz

tar -xvzf prometheus-2.51.1.linux-amd64.tar.gz

chown -R prometheus:prometheus /home/prometheus/prometheus-2.51.1.linux-amd64

cd /home/prometheus/prometheus-2.51.1.linux-amd64

./promtool check config prometheus.yml

sudo tee /etc/systemd/system/prometheus.service > /dev/null <<'EOF'
[Unit]
Description=Prometheus
Wants=network-online.target
After=network-online.target

[Service]
User=prometheus
Group=prometheus
Type=simple
Restart=on-failure
ExecStart=/home/prometheus/prometheus-2.51.1.linux-amd64/prometheus \
    --config.file=/home/prometheus/prometheus-2.51.1.linux-amd64/prometheus.yml

[Install]
WantedBy=multi-user.target
EOF

sudo systemctl daemon-reload
sudo systemctl enable prometheus
sudo systemctl start prometheus
sudo systemctl status prometheus

sudo ss -lntp | grep 9090
curl http://localhost:9090
